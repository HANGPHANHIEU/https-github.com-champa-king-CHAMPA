document.addEventListener('DOMContentLoaded', () => { // --- QUẢN LÝ TRẠNG THÁI NGƯỜI DÙNG TRONG LOCALSTORAGE --- window.getCurrentUser = function() { try { const user = localStorage.getItem('current_logged_user'); return user ? JSON.parse(user) : null; } catch (e) { return null; } }; window.setCurrentUser = function(user) { if (user) { localStorage.setItem('current_logged_user', JSON.stringify(user)); } else { localStorage.removeItem('current_logged_user'); } updateAuthButtonUI(); }; // --- INJECT MODAL ĐĂNG NHẬP / ĐĂNG KÝ VÀO #authModalContainer --- const authModalContainer = document.getElementById('authModalContainer'); if (authModalContainer) { authModalContainer.innerHTML = `

Đăng nhập tài khoản

Truy cập để đăng và quản lý việc làm dễ dàng

Chưa có tài khoản? Đăng ký ngay

Đã có tài khoản? Đăng nhập

`; } // --- XỬ LÝ GIAO DIỆN NÚT ĐĂNG NHẬP TRÊN HEADER --- function updateAuthButtonUI() { const btnLoginAccount = document.getElementById('btnLoginAccount'); const loginBtnText = document.getElementById('loginBtnText'); const currentUser = getCurrentUser(); if (btnLoginAccount && loginBtnText) { if (currentUser) { loginBtnText.textContent = currentUser.name.split(' ').pop(); // Hiển thị tên ngắn gọn btnLoginAccount.title = `Đang đăng nhập: ${currentUser.name} (${currentUser.phone})\nBấm để đăng xuất`; btnLoginAccount.classList.remove('bg-slate-100', 'text-slate-700'); btnLoginAccount.classList.add('bg-teal-50', 'text-teal-700', 'border', 'border-teal-200'); } else { loginBtnText.textContent = 'Đăng nhập'; btnLoginAccount.title = 'Bấm để đăng nhập'; btnLoginAccount.classList.remove('bg-teal-50', 'text-teal-700', 'border', 'border-teal-200'); btnLoginAccount.classList.add('bg-slate-100', 'text-slate-700'); } } } // --- SỰ KIỆN CLICK NÚT ĐĂNG NHẬP / ĐĂNG XUẤT --- const btnLoginAccount = document.getElementById('btnLoginAccount'); const authModal = document.getElementById('authModal'); const btnCloseAuthModal = document.getElementById('btnCloseAuthModal'); const loginForm = document.getElementById('loginForm'); const registerForm = document.getElementById('registerForm'); const authTitle = document.getElementById('authTitle'); const switchToRegister = document.getElementById('switchToRegister'); const switchToLogin = document.getElementById('switchToLogin'); if (btnLoginAccount) { btnLoginAccount.addEventListener('click', () => { const currentUser = getCurrentUser(); if (currentUser) { // Nếu đã đăng nhập, hỏi có muốn đăng xuất không if (confirm(`Bạn có muốn đăng xuất tài khoản (${currentUser.name}) không?`)) { setCurrentUser(null); if (typeof showToast === 'function') showToast('Đã đăng xuất tài khoản thành công!'); } } else { // Chưa đăng nhập -> Mở modal authModal?.classList.remove('hidden'); authModal?.classList.add('flex'); } }); } if (btnCloseAuthModal) { btnCloseAuthModal.addEventListener('click', () => { authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); }); } if (switchToRegister) { switchToRegister.addEventListener('click', (e) => { e.preventDefault(); loginForm?.classList.add('hidden'); registerForm?.classList.remove('hidden'); if (authTitle) authTitle.textContent = 'Đăng ký tài khoản'; }); } if (switchToLogin) { switchToLogin.addEventListener('click', (e) => { e.preventDefault(); registerForm?.classList.add('hidden'); loginForm?.classList.remove('hidden'); if (authTitle) authTitle.textContent = 'Đăng nhập tài khoản'; }); } // Xử lý Submit Form Đăng nhập if (loginForm) { loginForm.addEventListener('submit', (e) => { e.preventDefault(); const phone = document.getElementById('loginPhoneInput').value.trim(); const pass = document.getElementById('loginPasswordInput').value.trim(); // Lấy danh sách thành viên đã đăng ký trong localStorage hoặc tạo mẫu let users = []; try { users = JSON.parse(localStorage.getItem('app_registered_users')) || []; } catch (err) { users = []; } const found = users.find(u => u.phone === phone && u.password === pass); if (found) { setCurrentUser(found); authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); loginForm.reset(); if (typeof showToast === 'function') showToast(`Chào mừng trở lại, ${found.name}!`); } else { alert('Số điện thoại hoặc mật khẩu không chính xác! (Hoặc bạn chưa có tài khoản, vui lòng đăng ký).'); } }); } // Xử lý Submit Form Đăng ký if (registerForm) { registerForm.addEventListener('submit', (e) => { e.preventDefault(); const name = document.getElementById('regNameInput').value.trim(); const phone = document.getElementById('regPhoneInput').value.trim(); const password = document.getElementById('regPasswordInput').value.trim(); let users = []; try { users = JSON.parse(localStorage.getItem('app_registered_users')) || []; } catch (err) { users = []; } if (users.some(u => u.phone === phone)) { alert('Số điện thoại này đã được đăng ký trước đó!'); return; } const newUser = { id: Date.now(), name, phone, password }; users.push(newUser); localStorage.setItem('app_registered_users', JSON.stringify(users)); // Tự động đăng nhập luôn sau khi đăng ký thành công setCurrentUser(newUser); authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); registerForm.reset(); if (typeof showToast === 'function') showToast('Đăng ký thành viên thành công!'); }); } // Khởi chạy kiểm tra trạng thái lúc tải trang updateAuthButtonUI(); });
