// --- QUẢN LÝ TÀI KHOẢN VÀ PHIÊN ĐĂNG NHẬP (AUTH.JS) --- // Lấy tài khoản đang đăng nhập window.getCurrentUser = function() { try { const user = localStorage.getItem('current_logged_user'); return user ? JSON.parse(user) : null; } catch (e) { return null; } }; // Cập nhật tài khoản đang đăng nhập window.setCurrentUser = function(user) { if (user) { localStorage.setItem('current_logged_user', JSON.stringify(user)); } else { localStorage.removeItem('current_logged_user'); } if (typeof updateAuthButtonUI === 'function') updateAuthButtonUI(); if (typeof renderJobs === 'function') renderJobs(); }; document.addEventListener('DOMContentLoaded', () => { // Tự động inject Modal Đăng nhập / Đăng ký vào HTML const authModalContainer = document.getElementById('authModalContainer'); if (authModalContainer) { authModalContainer.innerHTML = `

Đăng nhập tài khoản

Đăng nhập để đăng việc & quản lý các bài đăng của bạn

Chưa có tài khoản? Đăng ký ngay

Đã có tài khoản? Đăng nhập

`; } // Cập nhật giao diện nút trên Header window.updateAuthButtonUI = function() { const btnLoginAccount = document.getElementById('btnLoginAccount'); const loginBtnText = document.getElementById('loginBtnText'); const currentUser = getCurrentUser(); if (btnLoginAccount && loginBtnText) { if (currentUser) { loginBtnText.textContent = currentUser.name.split(' ').pop(); btnLoginAccount.title = `Đang đăng nhập: ${currentUser.name} (${currentUser.phone})\nBấm để đăng xuất`; btnLoginAccount.className = "bg-teal-50 text-teal-700 border border-teal-200 font-medium px-3 py-2 rounded-xl text-xs flex items-center transition shadow-sm"; } else { loginBtnText.textContent = "Đăng nhập"; btnLoginAccount.title = "Đăng nhập tài khoản"; btnLoginAccount.className = "bg-slate-100 hover:bg-slate-200 text-slate-700 font-medium px-3 py-2 rounded-xl text-xs flex items-center transition shadow-sm"; } } }; // Sự kiện Click nút Đăng nhập / Đăng xuất const btnLoginAccount = document.getElementById('btnLoginAccount'); const authModal = document.getElementById('authModal'); const btnCloseAuthModal = document.getElementById('btnCloseAuthModal'); const loginForm = document.getElementById('loginForm'); const registerForm = document.getElementById('registerForm'); const authTitle = document.getElementById('authTitle'); const switchToRegister = document.getElementById('switchToRegister'); const switchToLogin = document.getElementById('switchToLogin'); if (btnLoginAccount) { btnLoginAccount.addEventListener('click', () => { const currentUser = getCurrentUser(); if (currentUser) { if (confirm(`Tài khoản: ${currentUser.name} (${currentUser.phone})\n\nBạn có muốn đăng xuất không?`)) { setCurrentUser(null); if (typeof showToast === 'function') showToast('Đã đăng xuất tài khoản.'); } } else { authModal?.classList.remove('hidden'); authModal?.classList.add('flex'); } }); } btnCloseAuthModal?.addEventListener('click', () => { authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); }); switchToRegister?.addEventListener('click', (e) => { e.preventDefault(); loginForm?.classList.add('hidden'); registerForm?.classList.remove('hidden'); if (authTitle) authTitle.textContent = 'Đăng ký tài khoản mới'; }); switchToLogin?.addEventListener('click', (e) => { e.preventDefault(); registerForm?.classList.add('hidden'); loginForm?.classList.remove('hidden'); if (authTitle) authTitle.textContent = 'Đăng nhập tài khoản'; }); // Submit Đăng nhập loginForm?.addEventListener('submit', (e) => { e.preventDefault(); const phone = document.getElementById('loginPhoneInput').value.trim(); const pass = document.getElementById('loginPasswordInput').value.trim(); const users = JSON.parse(localStorage.getItem('app_registered_users') || '[]'); const found = users.find(u => u.phone === phone && u.password === pass); if (found) { setCurrentUser(found); authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); loginForm.reset(); if (typeof showToast === 'function') showToast(`Xin chào ${found.name}! Bạn có thể quản lý bài đăng của mình.`); } else { alert('Số điện thoại hoặc mật khẩu không chính xác!'); } }); // Submit Đăng ký registerForm?.addEventListener('submit', (e) => { e.preventDefault(); const name = document.getElementById('regNameInput').value.trim(); const phone = document.getElementById('regPhoneInput').value.trim(); const password = document.getElementById('regPasswordInput').value.trim(); const users = JSON.parse(localStorage.getItem('app_registered_users') || '[]'); if (users.some(u => u.phone === phone)) { alert('Số điện thoại này đã được đăng ký thành viên!'); return; } const newUser = { id: Date.now(), name, phone, password }; users.push(newUser); localStorage.setItem('app_registered_users', JSON.stringify(users)); // Đăng nhập luôn sau khi tạo tài khoản thành công setCurrentUser(newUser); authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); registerForm.reset(); if (typeof showToast === 'function') showToast('Đăng ký tài khoản thành công!'); }); updateAuthButtonUI(); });
