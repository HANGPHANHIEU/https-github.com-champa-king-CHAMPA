// ========================================== // AUTH.JS: QUẢN LÝ DỮ LIỆU & GIAO DIỆN XÁC THỰC // ========================================== function getUsers() { try { const saved = localStorage.getItem('app_users'); return saved ? JSON.parse(saved) : []; } catch (e) { return []; } } function registerUser(name, phone, password) { const users = getUsers(); if (users.some(u => u.phone === phone)) { return { success: false, message: 'Số điện thoại này đã được đăng ký!' }; } const newUser = { id: Date.now(), name: name.trim(), phone: phone.trim(), password: password.trim() }; users.push(newUser); localStorage.setItem('app_users', JSON.stringify(users)); setCurrentUser(newUser); return { success: true, message: 'Đăng ký tài khoản thành công!' }; } function loginUser(phone, password) { const users = getUsers(); const user = users.find(u => u.phone === phone.trim() && u.password === password.trim()); if (!user) { return { success: false, message: 'Số điện thoại hoặc mật khẩu không chính xác!' }; } setCurrentUser(user); return { success: true, message: 'Đăng nhập thành công!' }; } function setCurrentUser(user) { localStorage.setItem('current_user', JSON.stringify(user)); } function getCurrentUser() { try { const saved = localStorage.getItem('current_user'); return saved ? JSON.parse(saved) : null; } catch (e) { return null; } } function logoutUser() { localStorage.removeItem('current_user'); } // --- TỰ ĐỘNG NẠP GIAO DIỆN auth.html VÀ GẮN SỰ KIỆN --- document.addEventListener('DOMContentLoaded', () => { // Tạo container nếu trong index.html chưa có if (!document.getElementById('authModalContainer')) { const container = document.createElement('div'); container.id = 'authModalContainer'; document.body.appendChild(container); } // Tải nội dung từ auth.html fetch('auth.html') .then(res => res.text()) .then(html => { document.getElementById('authModalContainer').innerHTML = html; initAuthUIEvents(); }) .catch(err => console.log('Không thể tải file auth.html:', err)); }); function initAuthUIEvents() { const btnLoginAccount = document.getElementById('btnLoginAccount'); const loginBtnText = document.getElementById('loginBtnText'); const authModal = document.getElementById('authModal'); const btnCloseAuthModal = document.getElementById('btnCloseAuthModal'); const loginForm = document.getElementById('loginForm'); const registerForm = document.getElementById('registerForm'); const authTitle = document.getElementById('authTitle'); const switchToRegister = document.getElementById('switchToRegister'); const switchToLogin = document.getElementById('switchToLogin'); function updateLoginButtonUI() { if (!loginBtnText) return; const user = getCurrentUser(); if (user) { loginBtnText.textContent = user.name; if (btnLoginAccount) btnLoginAccount.title = "Nhấn để đăng xuất"; } else { loginBtnText.textContent = "Đăng nhập"; if (btnLoginAccount) btnLoginAccount.title = "Đăng nhập tài khoản"; } } updateLoginButtonUI(); // Lắng nghe sự kiện đăng nhập thành công từ nơi khác (nếu cần) window.addEventListener('userStateChanged', updateLoginButtonUI); if (btnLoginAccount) { btnLoginAccount.addEventListener('click', () => { const user = getCurrentUser(); if (user) { if (confirm(`Bạn đang đăng nhập: ${user.name} (${user.phone}). Bạn có muốn đăng xuất không?`)) { logoutUser(); updateLoginButtonUI(); alert('Đã đăng xuất thành công.'); } } else { authModal?.classList.remove('hidden'); authModal?.classList.add('flex'); loginForm?.classList.remove('hidden'); registerForm?.classList.add('hidden'); if (authTitle) authTitle.textContent = 'Đăng nhập tài khoản'; } }); } btnCloseAuthModal?.addEventListener('click', () => { authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); }); switchToRegister?.addEventListener('click', (e) => { e.preventDefault(); loginForm?.classList.add('hidden'); registerForm?.classList.remove('hidden'); if (authTitle) authTitle.textContent = 'Đăng ký tài khoản'; }); switchToLogin?.addEventListener('click', (e) => { e.preventDefault(); registerForm?.classList.add('hidden'); loginForm?.classList.remove('hidden'); if (authTitle) authTitle.textContent = 'Đăng nhập tài khoản'; }); loginForm?.addEventListener('submit', (e) => { e.preventDefault(); const phone = document.getElementById('loginPhone').value.trim(); const pass = document.getElementById('loginPassword').value.trim(); const res = loginUser(phone, pass); alert(res.message); if (res.success) { authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); loginForm.reset(); updateLoginButtonUI(); } }); registerForm?.addEventListener('submit', (e) => { e.preventDefault(); const name = document.getElementById('regName').value.trim(); const phone = document.getElementById('regPhone').value.trim(); const pass = document.getElementById('regPassword').value.trim(); const res = registerUser(name, phone, pass); alert(res.message); if (res.success) { authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); registerForm.reset(); updateLoginButtonUI(); } }); }
