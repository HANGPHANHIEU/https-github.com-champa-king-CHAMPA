import { initTrashModal } from './trash-modal.js'; import { updateAuthUI } from './auth-modal.js'; // --- 1. ĐỊNH NGHĨA CÁC HÀM XỬ LÝ TOÀN CỤC ĐỂ GẮN VÀO NÚT BẤM --- // Xử lý tính năng Nhận việc window.handleTakeJob = (jobId) => { const isLoggedIn = localStorage.getItem('viecgap_logged_in') === 'true' || !!localStorage.getItem('user'); if (!isLoggedIn) { alert("Vui lòng đăng nhập tài khoản trước khi nhận việc!"); document.getElementById('authModal')?.classList.remove('hidden'); document.getElementById('authModal')?.classList.add('flex'); return; } const jobs = JSON.parse(localStorage.getItem('viecgap_jobs') || '[]'); const job = jobs.find(j => j.id === jobId); const currentUserPhone = localStorage.getItem('viecgap_user_phone'); if (job && job.phone === currentUserPhone) { alert("Bạn không thể tự nhận công việc do chính mình đăng!"); return; } if (confirm(`Bạn có chắc chắn muốn nhận công việc này không?`)) { alert("Nhận việc thành công! Chủ bài đăng sẽ liên hệ với bạn."); } }; // Xử lý tính năng Xóa việc (Chỉ chính chủ mới xóa được) window.handleDeleteJob = (jobId) => { const currentUserPhone = localStorage.getItem('viecgap_user_phone'); const jobs = JSON.parse(localStorage.getItem('viecgap_jobs') || '[]'); const job = jobs.find(j => j.id === jobId); if (!currentUserPhone) { alert('Vui lòng đăng nhập để thực hiện tính năng xóa!'); return; } if (job && job.phone !== currentUserPhone) { alert('Bạn không có quyền xóa công việc do người khác đăng!'); return; } if (confirm('Bạn có chắc chắn muốn xóa bài đăng này không?')) { const updatedJobs = jobs.filter(j => j.id !== jobId); localStorage.setItem('viecgap_jobs', JSON.stringify(updatedJobs)); // Đưa vào thùng rác if (job) { const deletedJobs = JSON.parse(localStorage.getItem('viecgap_deleted_jobs') || '[]'); deletedJobs.push(job); localStorage.setItem('viecgap_deleted_jobs', JSON.stringify(deletedJobs)); } alert('Đã xóa thành công!'); window.renderJobs(); // Cập nhật lại giao diện ngay lập tức } }; // --- 2. HÀM RENDER DANH SÁCH VIỆC LÀM --- window.renderJobs = () => { const jobListContainer = document.getElementById('jobList'); const jobCountElement = document.getElementById('jobCount'); if (!jobListContainer) return; const jobs = JSON.parse(localStorage.getItem('viecgap_jobs') || '[]'); const currentUserPhone = localStorage.getItem('viecgap_user_phone'); if (jobCountElement) { jobCountElement.innerText = `${jobs.length} việc phù hợp`; } if (jobs.length === 0) { jobListContainer.innerHTML = `
Chưa có công việc nào được đăng. Hãy đăng nhập và "Đăng việc" ngay!
`; return; } jobListContainer.innerHTML = jobs.map(job => { // Chỉ hiện nút Xóa nếu đúng là chính chủ đăng bài const isOwner = currentUserPhone && job.phone === currentUserPhone; return `
${job.category || 'Khác'} ${job.urgent ? ' Gấp' : ''}

${job.title}

${job.salary || 'Thỏa thuận'}
${job.location || 'Châu Thành'} ${job.createdAt || ''}
${isOwner ? ` ` : ''}
`; }).join(''); }; // --- 3. KHỞI TẠO CÁC SỰ KIỆN KHI DOM ĐÃ TẢI XONG --- document.addEventListener('DOMContentLoaded', () => { console.log("--- HỆ THỐNG 'VIỆC GẤP' ĐÃ KHỞI ĐỘNG ---"); initTrashModal(); if (typeof updateAuthUI === 'function') { updateAuthUI(); } window.renderJobs(); // Dọn dẹp modal bị kẹt document.querySelectorAll('[id*="Modal"], [id*="modal"]').forEach(modal => { modal.classList.add('hidden'); modal.classList.remove('flex'); }); // --- Modal Đăng nhập --- const authModal = document.getElementById('authModal'); const authForm = document.getElementById('authForm'); const resetForm = document.getElementById('resetForm'); const authModalTitle = document.getElementById('authModalTitle'); const authFooter = document.getElementById('authFooter'); const resetFooter = document.getElementById('resetFooter'); const loginBtnText = document.getElementById('loginBtnText'); const checkLoginStatus = () => { const isLoggedIn = localStorage.getItem('viecgap_logged_in') === 'true' || !!localStorage.getItem('user'); if (isLoggedIn && loginBtnText) { loginBtnText.innerText = "Đã đăng nhập"; } return isLoggedIn; }; checkLoginStatus(); const openAuthModal = () => { if (authModal) { authModal.classList.remove('hidden'); authModal.classList.add('flex'); authForm?.classList.remove('hidden'); resetForm?.classList.add('hidden'); authFooter?.classList.remove('hidden'); resetFooter?.classList.add('hidden'); if (authModalTitle) authModalTitle.innerText = "Đăng nhập tài khoản"; } }; const closeAuthModal = () => { authModal?.classList.add('hidden'); authModal?.classList.remove('flex'); authForm?.reset(); resetForm?.reset(); }; authForm?.addEventListener('submit', (e) => { e.preventDefault(); const phone = document.getElementById('loginPhone')?.value; if (phone) { localStorage.setItem('viecgap_logged_in', 'true'); localStorage.setItem('viecgap_user_phone', phone); if (loginBtnText) loginBtnText.innerText = "Đã đăng nhập"; alert('Đăng nhập thành công!'); closeAuthModal(); if (typeof updateAuthUI === 'function') updateAuthUI(); window.renderJobs(); } }); document.getElementById('btnLoginAccount')?.addEventListener('click', openAuthModal); document.getElementById('btnCloseAuthModal')?.addEventListener('click', closeAuthModal); document.getElementById('btnCancelAuthModal')?.addEventListener('click', closeAuthModal); document.getElementById('btnCancelReset')?.addEventListener('click', closeAuthModal); // --- Modal Đăng việc --- const postJobModal = document.getElementById('postJobModal'); const postJobForm = document.getElementById('postJobForm'); const openPostJob = () => { if (!checkLoginStatus()) { alert('Vui lòng đăng nhập trước khi đăng công việc mới!'); openAuthModal(); return; } postJobModal?.classList.remove('hidden'); postJobModal?.classList.add('flex'); }; const closePostJobModal = () => { postJobModal?.classList.add('hidden'); postJobModal?.classList.remove('flex'); postJobForm?.reset(); }; document.getElementById('btnPostJob')?.addEventListener('click', openPostJob); document.getElementById('btnFloatingPostJob')?.addEventListener('click', openPostJob); document.getElementById('btnCloseModal')?.addEventListener('click', closePostJobModal); document.getElementById('btnCancelModal')?.addEventListener('click', closePostJobModal); postJobForm?.addEventListener('submit', (e) => { e.preventDefault(); const currentUserPhone = localStorage.getItem('viecgap_user_phone') || 'anonymous'; const newJob = { id: 'job_' + Date.now(), title: document.getElementById('jobTitleInput')?.value || 'Công việc mới', location: document.getElementById('jobLocationInput')?.value || 'Châu Thành', category: document.getElementById('jobCategoryInput')?.value || 'delivery', salary: document.getElementById('jobSalaryInput')?.value || 'Thỏa thuận', urgent: document.getElementById('jobUrgentInput')?.checked || false, phone: currentUserPhone, createdAt: new Date().toLocaleDateString('vi-VN') }; let jobs = JSON.parse(localStorage.getItem('viecgap_jobs') || '[]'); jobs.unshift(newJob); localStorage.setItem('viecgap_jobs', JSON.stringify(jobs)); closePostJobModal(); window.renderJobs(); }); // Chuyển đổi qua lại giữa Đăng nhập và Đổi mật khẩu document.getElementById('btnToggleReset')?.addEventListener('click', () => { authForm?.classList.add('hidden'); authFooter?.classList.add('hidden'); resetForm?.classList.remove('hidden'); resetFooter?.classList.remove('hidden'); if (authModalTitle) authModalTitle.innerText = "Đổi mật khẩu tài khoản"; }); document.getElementById('btnBackToLogin')?.addEventListener('click', () => { resetForm?.classList.add('hidden'); resetFooter?.classList.add('hidden'); authForm?.classList.remove('hidden'); authFooter?.classList.remove('hidden'); if (authModalTitle) authModalTitle.innerText = "Đăng nhập tài khoản"; }); // --- Modal Tổ chức --- const orgModal = document.getElementById('orgModal'); const closeOrgModal = () => { orgModal?.classList.add('hidden'); orgModal?.classList.remove('flex'); }; document.getElementById('btnJoinOrg')?.addEventListener('click', () => { orgModal?.classList.remove('hidden'); orgModal?.classList.add('flex'); }); document.getElementById('btnCloseOrgModal')?.addEventListener('click', closeOrgModal); document.getElementById('btnCancelOrgModal')?.addEventListener('click', closeOrgModal); // --- Modal Cá nhân --- const workerModal = document.getElementById('workerModal'); const closeWorkerModal = () => { workerModal?.classList.add('hidden'); workerModal?.classList.remove('flex'); }; document.getElementById('btnRegisterWorker')?.addEventListener('click', () => { workerModal?.classList.remove('hidden'); workerModal?.classList.add('flex'); }); document.getElementById('btnCloseWorkerModal')?.addEventListener('click', closeWorkerModal); document.getElementById('btnCancelWorkerModal')?.addEventListener('click', closeWorkerModal); // --- Modal Thùng rác --- const trashModal = document.getElementById('trashModal'); document.getElementById('btnOpenTrash')?.addEventListener('click', () => { trashModal?.classList.remove('hidden'); trashModal?.classList.add('flex'); }); document.getElementById('btnCloseTrashModal')?.addEventListener('click', () => trashModal?.classList.add('hidden')); // --- Chuyển đổi Tab điều hướng dưới cùng --- document.querySelectorAll('.nav-btn').forEach(btn => { btn.addEventListener('click', () => { const tabId = btn.getAttribute('data-tab'); if (!tabId) return; document.querySelectorAll('.tab-content').forEach(t => { if (t.id === tabId) { t.classList.remove('hidden'); } else { t.classList.add('hidden'); } }); }); }); });
